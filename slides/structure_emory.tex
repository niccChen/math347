\section{Introduction and Motivation}
% ============================================================================

\begin{frame}{Why Low-Rank Factorization?}
    
    \textbf{Real-World Problem:}
    \begin{itemize}
        \item High-dimensional data with missing or noisy entries
        \item User-item ratings (Netflix recommender systems)
        \item Image compression and denoising
        \item Latent feature discovery in scientific data
    \end{itemize}
    
    \vspace{0.4cm}
    
    \textbf{Key Insight:} Many real datasets have underlying \textbf{low-rank structure}
    \begin{itemize}
        \item User preferences: $\sim 5$--$10$ latent factors
        \item Images: compressible via wavelets (low-rank in transform domain)
        \item Parameter efficiency: $r(m + n)$ factors vs.\ $mn$ entries
        \item Example: $1000 \times 1700$ matrix, $r=20$ $\Rightarrow$ $\sim 10\times$ compression
    \end{itemize}
    
    \vspace{0.4cm}
    
    \textbf{Solution:} Low-rank factorization captures signal while discarding noise
\end{frame}

\begin{frame}{Project Overview and Roadmap}
    
    \textbf{Core Problem:} Compare three optimization algorithms for regularized matrix factorization
    
    \vspace{0.2cm}
    
    $$\min_{U, V} f(U, V) = \frac{1}{2}\|UV^T - M\|_F^2 + \frac{\lambda}{2}(\|U\|_F^2 + \|V\|_F^2)$$
    
    \vspace{0.4cm}
    
    \textbf{Three Optimization Methods:}
    \begin{itemize}
        \item \textcolor{emoryblue}{\textbf{Gradient Descent + Armijo Backtracking}} --- Simple, stable, pedagogical
        \item \textcolor{emoryblue}{\textbf{L-BFGS}} --- Quasi-Newton, fast superlinear convergence
        \item \textcolor{emoryblue}{\textbf{Alternating Least Squares (ALS)}} --- Block coordinate descent, ideal for sparse data
    \end{itemize}
    
    \vspace{0.4cm}
    
    \textbf{Three Datasets:}
    \begin{itemize}
        \item \textcolor{emoryblue}{\textbf{Synthetic}} --- $500 \times 200$, rank-5, controlled noise
        \item \textcolor{emoryblue}{\textbf{MovieLens 100K}} --- $943 \times 1682$ users $\times$ movies, 93.7\% sparse
        \item \textcolor{emoryblue}{\textbf{Superconductivity}} --- $21263 \times 81$ compounds $\times$ features, dense
    \end{itemize}
\end{frame}

\begin{frame}{Problem Formulation and Properties}
    
    \textbf{Regularized Low-Rank Matrix Factorization:}
    
    \begin{equation}
        \min_{U \in \mathbb{R}^{m \times r}, V \in \mathbb{R}^{n \times r}} 
        f(U, V) = \frac{1}{2}\|UV^T - M\|_F^2 + \frac{\lambda}{2}(\|U\|_F^2 + \|V\|_F^2)
    \end{equation}
    
    \vspace{0.3cm}
    
    \textbf{Problem Properties:}
    \begin{itemize}
        \item \textcolor{emoryblue}{\textbf{Non-convex}} in $(U, V)$ jointly, but \textbf{biconvex} separately
        \item \textcolor{emoryblue}{\textbf{L-smooth}}: Lipschitz continuous gradient enables convergence guarantees
        \item \textcolor{emoryblue}{\textbf{Multiple stationary points}}: Local minima exist but often similar quality
        \item Smoothness constant: $L \leq \sigma_{\max}(M)^2 + \lambda$
    \end{itemize}
    
    \vspace{0.3cm}
    
    \textbf{Regularization Trade-off:}
    \begin{itemize}
        \item $\lambda \to 0$: Fits noise, high test error (overfitting)
        \item $\lambda$ optimal: Balances fitting signal and regularization (sweet spot)
        \item $\lambda \to \infty$: Underfitting, poor reconstruction
    \end{itemize}
\end{frame}

\section{Optimization Methods}
% ============================================================================

\begin{frame}{Method 1: Gradient Descent with Armijo Backtracking}
    
    \textbf{Update Rule:} Gradient descent with line search
    
    $$U_{k+1} = U_k - \alpha_k \nabla_U f(U_k, V_k), \quad 
    V_{k+1} = V_k - \alpha_k \nabla_V f(U_k, V_k)$$
    
    \vspace{0.3cm}
    
    \textbf{Gradient Formulation (Dense):}
    $$\nabla_U f = (UV^T - M)V + \lambda U, \quad
    \nabla_V f = (UV^T - M)^T U + \lambda V$$
    
    \vspace{0.3cm}
    
    \textbf{Armijo Line Search:} Backtracking until sufficient decrease
    \begin{itemize}
        \item Start with $\alpha = 1$
        \item If condition fails: $\alpha \leftarrow 0.5 \alpha$ (halve step size)
        \item Continue until: $f(\theta - \alpha \nabla f) \leq f(\theta) - c \alpha \|\nabla f\|^2$
        \item Constant: $c = 10^{-4}$ (default)
    \end{itemize}
    
    \vspace{0.3cm}
    
    \textbf{Convergence:} Linear convergence to stationary point
\end{frame}

\begin{frame}{Method 2: L-BFGS (Limited-memory BFGS)}
    
    \textbf{Quasi-Newton Method:} Approximate Hessian with limited history
    
    \vspace{0.3cm}
    
    \textbf{Key Features:}
    \begin{itemize}
        \item Maintains $m$ recent gradient/step pairs (typically $m = 10$)
        \item Approximates Hessian: $B_k \approx \nabla^2 f$
        \item Update: $x_{k+1} = x_k - \alpha_k B_k^{-1} \nabla f(x_k)$
        \item Automatic backtracking and Hessian reconstruction
    \end{itemize}
    
    \vspace{0.3cm}
    
    \textbf{Implementation (scipy):}
    \begin{itemize}
        \item Pack factors: $x = [U; V] \in \mathbb{R}^{(m+n)r}$
        \item Call \texttt{scipy.optimize.minimize(method='L-BFGS-B')}
        \item Black-box optimizer with automatic tuning
    \end{itemize}
    
    \vspace{0.3cm}
    
    \textbf{Convergence:} Superlinear convergence
    \begin{itemize}
        \item Practical: $2$--$3\times$ fewer iterations than GD
        \item Rate: $\|x_k - x^*\| = o(\|x_{k-1} - x^*\|)$
    \end{itemize}
\end{frame}

\begin{frame}{Method 3: Alternating Least Squares (ALS)}
    
    \textbf{Block Coordinate Descent:} Optimize $U$ and $V$ alternately
    
    \vspace{0.3cm}
    
    \textbf{Fix $V$, optimize $U$ (closed-form solution):}
    
    $$\min_U \frac{1}{2}\|UV^T - M\|_F^2 + \frac{\lambda}{2}\|U\|_F^2 \quad \Rightarrow \quad 
    U = MV(V^TV + \lambda I)^{-1}$$
    
    \vspace{0.3cm}
    
    \textbf{Fix $U$, optimize $V$ (symmetric form):}
    
    $$V = M^T U(U^T U + \lambda I)^{-1}$$
    
    \vspace{0.3cm}
    
    \textbf{Sparse Implementation:} Row-by-row updates
    \begin{itemize}
        \item For each row $i$, solve ridge regression using only observed entries
        \item Natural handling of missing data (MovieLens-type problems)
        \item Computational complexity: $O(\text{nnz} \cdot r^2)$ per iteration
    \end{itemize}
    
    \vspace{0.3cm}
    
    \textbf{Convergence:} Sublinear descent, guaranteed to stationary point via biconvexity
\end{frame}

\begin{frame}{Method Comparison: Complexity and Convergence}
    
    \begin{table}[h]
    \centering
    \renewcommand{\arraystretch}{1.2}
    \small
    \begin{tabular}{|l|c|c|c|}
    \hline
    \textbf{Method} & \textbf{Convergence} & \textbf{Per-Iteration Cost} & \textbf{Sparse Suitability} \\
    \hline
    Gradient Descent & Linear & $O(mnr + r^2 m + r^2 n)$ & Limited \\
    \hline
    L-BFGS & Superlinear & $O((m+n)r)$ & Possible \\
    \hline
    ALS & Sublinear & $O(\text{nnz} \cdot r^2)$ & \textcolor{emoryblue}{\textbf{Ideal}} \\
    \hline
    \end{tabular}
    \end{table}
    
    \vspace{0.4cm}
    
    \textbf{Practical Trade-offs:}
    \begin{itemize}
        \item \textcolor{emoryblue}{\textbf{For sparse data}}: ALS wins (fewest per-iteration operations)
        \item \textcolor{emoryblue}{\textbf{For dense data}}: L-BFGS wins (fewest iterations overall)
        \item \textcolor{emoryblue}{\textbf{For stability}}: GD most reliable (no hyperparameter tuning)
    \end{itemize}
\end{frame}

    \textbf{Synthetic Data:}
    \begin{itemize}
        \item Generation: $M = U_{\text{true}} V_{\text{true}}^T + \sigma \cdot \mathcal{N}(0,1)$
        \item Parameters: rank 5, $\sigma = 0.1$
        \item Advantage: know true solution $\Rightarrow$ measure recovery error directly
    \end{itemize}
    
    \vspace{0.2cm}
    
    \textbf{MovieLens 100K (Real Sparse):}
    \begin{itemize}
        \item 943 users, 1682 movies, $\sim 100$K ratings (5-star scale)
        \item Train/test split: evaluate RMSE on held-out ratings
    \end{itemize}
\end{frame}

\begin{frame}{Experimental Design: Five Core Experiments}
    
    \textbf{Experiment 1: Method Comparison Across Datasets}
    \begin{itemize}
        \item Compare GD, L-BFGS, ALS on all three datasets
        \item Metrics: iterations, wall-clock time, final objective, reconstruction error
    \end{itemize}
    
    \vspace{0.3cm}
    
    \textbf{Experiment 2: Initialization Sensitivity}
    \begin{itemize}
        \item Test random vs.\ SVD initialization
        \item Question: Can poor initialization trap us in bad local minima?
    \end{itemize}
    
    \vspace{0.3cm}
    
    \textbf{Experiment 3: Rank Selection}
    \begin{itemize}
        \item Vary rank: study underfitting vs.\ overfitting trade-off
        \item Expected U-shaped error curve
    \end{itemize}
    
    \vspace{0.3cm}
    
    \textbf{Experiment 4: Regularization Sensitivity}
    \begin{itemize}
        \item Grid search over $\lambda \in [10^{-4}, 1.0]$
        \item How does $\lambda$ affect recovery under noise?
    \end{itemize}
    
    \vspace{0.3cm}
    
    \textbf{Experiment 5: Optimization Landscape}
    \begin{itemize}
        \item Visualize nonconvex effects via interpolation between solutions
        \item Assess local minima quality variance
    \end{itemize}
\end{frame}

\section{Key Findings}
% ============================================================================

\begin{frame}{Finding 1: Nonconvexity Doesn't Kill Optimization}
    
    \textbf{Main Result:} All three methods converge reliably despite nonconvexity
    
    \vspace{0.3cm}
    
    \textbf{Evidence:}
    \begin{itemize}
        \item All three converge to (possibly different) local minima
        \item With good initialization, convergence is fast
        \item ALS especially robust via block coordinate descent
        \item Regularization ensures well-conditioned problem
    \end{itemize}
    
    \vspace{0.3cm}
    
    \textbf{Implication:} Nonconvexity is \textbf{not} a practical barrier for this problem
    \begin{itemize}
        \item Focus shifts to: initialization quality, rank selection, regularization tuning
        \item Multiple local minima exist but are often similar quality
    \end{itemize}
    
    \vspace{0.3cm}
    
    \textbf{Practical Lesson:} For well-regularized problems with reasonable initialization,
    \\ nonconvexity is manageable
\end{frame}

\begin{frame}{Finding 2: Rank and Initialization Matter Most}
    
    \textbf{Critical Factors (in order of importance):}
    
    \vspace{0.3cm}
    
    \begin{enumerate}
        \item \textcolor{emoryblue}{\textbf{Rank Selection}}
        \begin{itemize}
            \item Underestimating: Underfitting bias dominates
            \item Overestimating: Overfitting risk (especially sparse data)
            \item Optimal rank matches intrinsic signal dimension
        \end{itemize}
        
        \vspace{0.2cm}
        
        \item \textcolor{emoryblue}{\textbf{Initialization Quality}}
        \begin{itemize}
            \item SVD initialization $\gg$ random initialization
            \item SVD provides 5--10$\times$ faster convergence
            \item Initialization choice $>$ algorithm choice for solution quality
        \end{itemize}
        
        \vspace{0.2cm}
        
        \item \textcolor{emoryblue}{\textbf{Regularization}}
        \begin{itemize}
            \item MovieLens: $\lambda \in [0.1, 1.0]$ works well
            \item Dense data: less sensitive to $\lambda$ choice
            \item Prevents noise fitting and ensures stability
        \end{itemize}
    \end{enumerate}
\end{frame}

\begin{frame}{Finding 3: Algorithm Trade-offs}
    
    \begin{columns}[T]
    \column{0.32\textwidth}
    
    \textbf{\textcolor{emoryblue}{Gradient Descent}}
    \begin{itemize}
        \item [\checkmark] Most stable
        \item [\checkmark] Easiest to debug
        \item [\checkmark] Best pedagogical
        \item [\times] Slowest convergence
    \end{itemize}
    
    \column{0.32\textwidth}
    
    \textbf{\textcolor{emoryblue}{L-BFGS}}
    \begin{itemize}
        \item [\checkmark] Fewest iterations
        \item [\checkmark] Good for dense data
        \item [\checkmark] Robust convergence
        \item [\times] More complex
    \end{itemize}
    
    \column{0.32\textwidth}
    
    \textbf{\textcolor{emoryblue}{ALS}}
    \begin{itemize}
        \item [\checkmark] Best for sparse
        \item [\checkmark] Fastest per iter
        \item [\checkmark] Natural for missing data
        \item [\times] May converge differently
    \end{itemize}
    \end{columns}
    
    \vspace{0.4cm}
    
    \textbf{Practical Guidance:}
    \begin{itemize}
        \item \textcolor{emoryblue}{\textbf{Recommender systems}}: ALS with SVD initialization
        \item \textcolor{emoryblue}{\textbf{Image/dense tasks}}: L-BFGS with SVD initialization
        \item \textcolor{emoryblue}{\textbf{Research/teaching}}: GD for interpretability
    \end{itemize}
\end{frame}

\section{Conclusions and Recommendations}
% ============================================================================

\begin{frame}{Practical Recommendations}
    
    \textbf{For Sparse Data (Recommender Systems):}
    \begin{itemize}
        \item Use \textcolor{emoryblue}{\textbf{ALS}} with SVD initialization
        \item Regularization: $\lambda \in [0.1, 1.0]$
        \item Rank: 20--50 for typical collaborative filtering
        \item Expect: fast convergence, natural missing-data handling
    \end{itemize}
    
    \vspace{0.3cm}
    
    \textbf{For Dense Data (Image Compression):}
    \begin{itemize}
        \item Use \textcolor{emoryblue}{\textbf{L-BFGS}} (fewer iterations, fast wall-clock time)
        \item SVD initialization essential
        \item Rank determined by compression requirements
        \item Expect: superlinear convergence
    \end{itemize}
    
    \vspace{0.3cm}
    
    \textbf{Managing Nonconvexity:}
    \begin{itemize}
        \item Initialize well: SVD beats random by large margin
        \item Regularize appropriately: prevents noise fitting
        \item Well-regularized problem $\Rightarrow$ few local minima of similar quality
        \item Multiple runs less necessary if initialized well
    \end{itemize}
\end{frame}

\begin{frame}{Main Conclusions}
    
    \textbf{1. Low-Rank Factorization is Powerful}
    \begin{itemize}
        \item Applicable across recommender systems, image processing, scientific computing
        \item Compression and denoising benefits are substantial
    \end{itemize}
    
    \vspace{0.3cm}
    
    \textbf{2. Nonconvex $\neq$ Unsolvable}
    \begin{itemize}
        \item Good initialization + regularization $\Rightarrow$ reliable optimization
        \item Multiple local minima exist but are similar quality
    \end{itemize}
    
    \vspace{0.3cm}
    
    \textbf{3. Algorithm Choice Depends on Data Structure}
    \begin{itemize}
        \item Sparse: \textbf{ALS} best
        \item Dense: \textbf{L-BFGS} best
        \item Stability: \textbf{GD} best
    \end{itemize}
    
    \vspace{0.3cm}
    
    \textbf{4. Open Questions for Future Work}
    \begin{itemize}
        \item Automatic rank selection (cross-validation, information criteria)
        \item Theoretical guarantees avoiding local minima
        \item Scalability to billion-scale matrices (stochastic, distributed methods)
    \end{itemize}
\end{frame}
