\section{Introduction}

\begin{frame}{Regularized Low-Rank Matrix Factorization}
    Low-rank matrix factorization is a technique that decomposes a large matrix $M \in \Real^{m \times n}$ into a product of two smaller matrices:
    \[M \approx U V^T\]

where $U \in \Real^{m \times r}$ and $V \in \Real^{n \times r}$, with $r \ll \min(m, n)$ being the rank of the factorization.

\vspace{0.25cm}

\textbf{Core Problem:}
    \[
    \min_{U, V} 
    f(U, V) = \frac{1}{2}\|UV^T - M\|_F^2 + \frac{\lambda}{2}(\|U\|_F^2 + \|V\|_F^2)
    \]

    where:
\begin{itemize}
    \item $U \in \Real^{m \times r}$ and $V \in \Real^{n \times r}$ are factor matrices; $M \in \Real^{m \times n}$ is the observed matrix 
    \item $\|\cdot\|_F$ denotes the Frobenius norm: $\|A\|_F = \sqrt{\sum_{i} \sum_j a_{ij}^2}$
    \item First term penalizes the difference between $UV^T$ and $M$
    \item Second term involves the ridge regularization parameter $\lambda \geq 0$
\end{itemize}
\end{frame}

\begin{frame}{Problem Formulation}

    \textbf{Main Question:} How do GD, L-BFGS, and ALS behave on dense vs.\ sparse regularized matrix factorization?

\vspace{0.25cm}

\textbf{Optimization Methods: }
\begin{itemize}
  \item \textbf{Gradient Descent (GD) + Armijo backtracking}: simple first-order baseline, stable but slow near minima 
  \item \textbf{L-BFGS}: quasi-Newton method with limited memory, fast local superlinear convergence
  \item \textbf{Alternating Least Squares (ALS)}:  block coordinate descent with closed-form updates, natural for observed-entry sparse data
\end{itemize}

\vspace{0.2cm}

\textbf{Key Difference:} Dense $M$ (full $\Omega$) vs.\ sparse $M$ (few observed entries) changes gradient computation costs and convergence behavior.

\end{frame}

\begin{frame}{Problem Properties}

\textbf{Problem Characteristics:}
\begin{itemize}
    \item \textbf{Non-convexity}: Objective function is jointly non-convex in $(U, V)$; but convex in $U$ and $V$ individually when fixing the other.
    \item \textbf{L-smoothness}: Gradient is Lipschitz continuous, guarantee convergence for gradient-based methods under appropriate step sizes.
    \item \textbf{Multiple stationary points}: Local minima exist, but in practice they often yield comparable reconstruction quality.
    \item \textbf{Lipschitz constant bound}: $L \leq \sigma_{\max}(M)^2 + \lambda$ ($\sigma_{\max}(M)$ is the largest singular value).
\end{itemize}

\vspace{0.25cm}

    \textbf{Regularization Trade-off:}
    \begin{itemize}
        \item $\lambda \to 0$: The model fits noise, leaing to high test error (overfitting)
        \item $\lambda$ optimal: Balances fitting signal and regularization, achieving best generalization 
        \item $\lambda \to \infty$: Excessive regularization causes poor reconstruction (underfitting)
    \end{itemize}
\end{frame}

\begin{frame}{Dense and Sparse Formulations}

\textbf{Dense objective (Superconductivity, Synthetic-dense):}
\[
f(U,V)
= \frac{1}{2}\lVert UV^T - M\rVert_F^2
+ \frac{\lambda}{2}\bigl(\lVert U\rVert_F^2 + \lVert V\rVert_F^2\bigr)
\]

\vspace{0.3cm}

\textbf{Sparse objective (MovieLens 100K, Synthetic-sparse):}
\[
f(U,V)
= \frac{1}{2}\sum_{(i,j)\in\Omega}\bigl(u_i^T v_j - M_{ij}\bigr)^2
+ \frac{\lambda}{2}\bigl(\lVert U\rVert_F^2 + \lVert V\rVert_F^2\bigr)
\]

\vspace{0.2cm}
\footnotesize
For MovieLens, the final tuned sparse model additionally includes user/item bias terms in the prediction function.

\end{frame}

\section{Optimization Methods}
% ============================================================================

\begin{frame}{Method 1: Gradient Descent with Armijo Backtracking}

\textbf{Update Rule:} Gradient descent with line search

$$U_{k+1} = U_k - \alpha_k \nabla_U f(U_k, V_k), \quad 
V_{k+1} = V_k - \alpha_k \nabla_V f(U_k, V_k)$$

\vspace{0.3cm}

\textbf{Gradient Formulation (Dense):}
$$\nabla_U f = (UV^T - M)V + \lambda U, \quad
\nabla_V f = (UV^T - M)^T U + \lambda V$$

\vspace{0.3cm}
\textbf{Gradient Formulation (Sparse):}
\[
\nabla_{u_i} f
=
\sum_{j:(i,j)\in\Omega}\left(u_i^\top v_j - M_{ij}\right)v_j + \lambda u_i,
\qquad
\nabla_{v_j} f
=
\sum_{i:(i,j)\in\Omega}\left(u_i^\top v_j - M_{ij}\right)u_i + \lambda v_j
\]


\vspace{0.3cm}

\textbf{Takeaway:} GD is the simplest and most stable baseline.
\end{frame}

\begin{frame}{Method 1: Gradient Descent with Armijo Backtracking}

\textbf{Armijo Line Search:} Backtracking until sufficient decrease

\begin{itemize}
    \item Start with $\alpha = 1$
    \item If condition fails: $\alpha \leftarrow \beta \cdot \alpha$ (typical $\beta = 0.5$)
    \item Continue until: $f(\theta - \alpha \nabla f) \leq f(\theta) - c \alpha \|\nabla f\|^2$
    \item Constant: $c = 10^{-4}$ (default)
\end{itemize}

\vspace{0.4cm}

\textbf{Convergence Behavior:}

\begin{itemize}
    \item Armijo backtracking guarantees sufficient decrease in the objective at each accepted step.
    \item For our non-convex matrix factorization problem, GD typically converges to a stationary point rather than guaranteeing a global optimum.
    \item In this project, GD is mainly used as a stable and interpretable baseline across different datasets.
\end{itemize}

\vspace{0.4cm}

\textbf{Takeaway:} GD with Armijo backtracking provides monotone objective decrease and adaptive step-size control, making it a reliable baseline for comparing dense and sparse data experiments.

\end{frame}

\begin{frame}{Method 2: L-BFGS (Quasi-Newton Method)}
    \textbf{Quasi-Newton Idea:} Approximate the Hessian (or its inverse) using past gradient differences, avoiding explicit computation of \(\nabla^2 f\).

    \vspace{0.2cm}
    \textbf{Update Rule:}
    \[
    x_{k+1} = x_k - \alpha_k H_k \nabla f(x_k),
    \qquad H_k \approx \bigl(\nabla^2 f(x_k)\bigr)^{-1}
    \]

    \vspace{0.2cm}
    \textbf{Curvature Pairs:} 
    \[
    s_k = x_{k+1} - x_k, \qquad
    y_k = \nabla f(x_{k+1}) - \nabla f(x_k)
    \]
    These capture local curvature information along the search direction.

\vspace{0.2cm}
    \textbf{Limited Memory:}
    \begin{itemize}
        \item Stores only the most recent \(m\) curvature pairs \(\{(s_i, y_i)\}\) (typically \(m = 5\)–\(20\)).
        \item Discards older pairs – adapts to changing curvature.
    \end{itemize}
\end{frame}










\begin{frame}{Method 2: L-BFGS (Quasi-Newton Method)}

    \vspace{0.2cm}
    \textbf{Two-Loop Recursion:}
    \begin{itemize}
        \item Computes the search direction \(p_k = -H_k \nabla f(x_k)\) without forming \(H_k\) explicitly.
        \item Uses the stored pairs to sequentially update an initial diagonal approximation.
    \end{itemize}

    \vspace{0.2cm}
    \textbf{Line Search:}
    \begin{itemize}
        \item Step length \(\alpha_k\) is chosen via backtracking conditions to ensure sufficient decrease.
        \item Guarantees stability and convergence for non-convex problems.
    \end{itemize}

    \vspace{0.2cm}
    \textbf{Takeaway:} L-BFGS efficiently exploits curvature information from a short history, making it suitable for large-scale optimization where storing the full Hessian is infeasible.
\end{frame}




\begin{frame}{Method 3: Alternating Least Squares (ALS)}

\textbf{Core Idea:} Alternately fix one factor matrix and solve for the other exactly (closed form).

\vspace{0.2cm}
\textbf{Fix $V$, optimize $U$:}
\[
\min_U \frac{1}{2}\|UV^T - M\|_F^2 + \frac{\lambda}{2}\|U\|_F^2
\]
Solution:
\[
U = M V (V^T V + \lambda I)^{-1}
\]

\vspace{0.2cm}
\textbf{Fix $U$, optimize $V$:}
\[
\min_V \frac{1}{2}\|UV^T - M\|_F^2 + \frac{\lambda}{2}\|V\|_F^2
\]
Solution:
\[
V = M^T U (U^T U + \lambda I)^{-1}
\]

\vspace{0.2cm}
\textbf{Iteration:} Repeat the two updates until convergence.  

\end{frame}






\begin{frame}{Method 3: Alternating Least Squares (ALS)}
\textbf{Sparse Data Handling (partially observed \(M\))}

\[
\min_{U,V}
\frac12\sum_{(i,j)\in\Omega}\left(M_{ij}-u_i^\top v_j\right)^2
+\frac{\lambda}{2}\left(\|U\|_F^2+\|V\|_F^2\right)
\]

\begin{itemize}
    \item For sparse data, we fit only the observed entries \((i,j)\in\Omega\).
    \item Fix \(V\), and update each row factor \(u_i\) separately.
    \item Let
    \[
    \Omega_i=\{j:(i,j)\in\Omega\}
    \]
    be the observed columns in row \(i\).
    \item Then \(u_i\) is updated by solving
    \[
    \min_{u_i}\;
    \frac12\sum_{j\in\Omega_i}\left(M_{ij}-u_i^\top v_j\right)^2
    +\frac{\lambda}{2}\|u_i\|_2^2.
    \]
\end{itemize}
\end{frame}


\begin{frame}{Method 3: ALS for Sparse Data (continued)}
\textbf{Closed-form row update:}
\[
u_i=
\left(V_{\Omega_i}^\top V_{\Omega_i}+\lambda I\right)^{-1}
V_{\Omega_i}^\top M_{i,\Omega_i}.
\]

\vspace{0.2cm}
\textbf{Similarly, for each column \(j\):}
\[
\Omega^j=\{i:(i,j)\in\Omega\}
\]
and
\[
v_j=
\left(U_{\Omega^j}^\top U_{\Omega^j}+\lambda I\right)^{-1}
U_{\Omega^j}^\top M_{\Omega^j,j}.
\]

\begin{itemize}
    \item Missing entries are ignored directly, so no imputation is needed.
    \item ALS alternates between row-wise and column-wise updates until convergence.
\end{itemize}

\textbf{Takeaway:}
ALS naturally handles sparse matrix completion by using only observed entries.
\end{frame}


\begin{frame}{Method 3: Alternating Least Squares (ALS)}


\vspace{0.2cm}
\textbf{Convergence:}
\begin{itemize}
    \item ALS monotonically decreases the objective (block-coordinate descent).
    \item Converges to a stationary point (often a local minimum).
    \item Can be sensitive to initialization.
\end{itemize}

\vspace{0.2cm}
\textbf{Takeaway:} ALS is a robust block-coordinate method for sparse matrix completion, with closed-form updates and quadratic convergence in each subproblem.
\end{frame}

\end{frame}

\begin{frame}{Observed Trade-offs: Experimental Summary}

\textbf{1. GD:}
    \begin{itemize}
        \item Stable across all problems (baseline), simple to implement with Armijo backtracking
        \item Slow to reach high accuracy due to sublinear convergence
    \end{itemize}

        \vspace{0.15cm}
        
\textbf{2. L-BFGS:} 
    \begin{itemize}
        \item Fastest on dense synthetic problems, but performance slightly degrades on sparse data
        \item Benefits from curvature approximation without storing full Hessian
    \end{itemize}
    
    \vspace{0.15cm}
    
    \textbf{3. ALS:}
    \begin{itemize}
        \item Monotonic improvement (guaranteed decrease), effective for sparse matrix
        \item Handles missing entries without imputation or dense matrix operations
        \item $O(r^2)$ factor in per-iteration cost hurts for large rank $r > 100$
    \end{itemize}

\vspace{0.25cm}
\textbf{Insight:} No single method dominates, depending on rank and sparsity.

\end{frame}

\begin{itemize}
  \item Same rank \& initialization across methods \item Records objective value, iterations, and runtime for each method 
  \item Synthetic datasets are compared to the clean ground-truth matrix
\end{itemize}

\vspace{0.2cm}

\textbf{Dataset-specific metrics}
\begin{itemize}
    \item \textbf{Synthetic}: objective + recovery error
    \item \textbf{MovieLens}: train/test RMSE
    \item \textbf{Superconductivity}: objective + reconstruction error
\end{itemize}

\vspace{0.2cm}

\textbf{Hyperparameters}
\begin{itemize}
  \item Dense and synthetic experiments use $\lambda = 0.01$
  \item MovieLens tuned values: $r = 3$, \ $\lambda = 2.0$, \ $\lambda_{\text{bias}} = 0.1$
\end{itemize}
\end{frame}

\begin{frame}{Special Case: MovieLens Dataset}

\textbf{Extended Prediction Function:}
\[
\hat r_{ij} = \mu + b_i + c_j + u_i^\top v_j
\]

\textbf{Regularized Objective:}
\[
\min \; \frac{1}{2}\sum_{(i,j)\in\Omega}(\hat r_{ij}-r_{ij})^2
+ \frac{\lambda}{2}(\|U\|_F^2+\|V\|_F^2)
+ \frac{\lambda_{\text{bias}}}{2}(\|b\|_2^2+\|c\|_2^2)
\]

\textbf{Interpretation:}
\begin{itemize}
    \item $\mu + b_i + c_j$: interpretable baseline (user/item biases)
    \item $u_i^\top v_j$: collaborative filtering residual
    \item Standard matrix factorization recovered if $\lambda_{\text{bias}} \to \infty$ (biases $\to 0$)
\end{itemize}

\end{frame}

\begin{frame}{Appendix: GD Convergence ($L$-smooth case)}

\textbf{Assumption:} $f$ is $L$-smooth (gradient Lipschitz):
\[
\|\nabla f(x) - \nabla f(y)\| \leq L \|x - y\|
\]

\textbf{Descent Lemma:} For any step size $\alpha_k$,
\[
f(x_{k+1}) \leq f(x_k) - \alpha_k \|\nabla f(x_k)\|^2 + \frac{L}{2}\alpha_k^2 \|\nabla f(x_k)\|^2
\]

\textbf{Choose step size:} $\alpha_k = \frac{1}{L}$. Then the quadratic term cancels:
\[
f(x_{k+1}) \leq f(x_k) - \frac{1}{2L} \|\nabla f(x_k)\|^2
\]

\textbf{Convergence rate (sublinear):} Summing telescopically,
\[
\min_{k \le T} \|\nabla f(x_k)\|^2 \leq \frac{2L\big(f(x_0) - f^*\big)}{T} = O\left(\frac{1}{T}\right)
\]

Sublinear convergence to a stationary point; possibly slow near flat regions.
\end{frame}

\begin{frame}{Appendix: L-BFGS Convergence}

\textbf{Assumptions:}
\begin{itemize}
    \item $f$ twice continuously differentiable
    \item $\nabla^2 f(x^*)$ positive definite (local strong convexity)
\end{itemize}

\textbf{Local convergence rate:}
\[
\|x_{k+1} - x^*\| \leq C \|x_k - x^*\|^{1+\epsilon}, \quad \epsilon > 0
\]
i.e., \textbf{superlinear} (faster than linear, slower than quadratic).

\textbf{Requirements:}
\begin{itemize}
    \item Good initialization (within neighborhood of $x^*$)
    \item Limited memory $m$ sufficiently large (but $m \ge 5$ often enough)
\end{itemize}

Rapid local convergence once curvature is well approximated, without storing the full Hessian.

\end{frame}

\begin{frame}{Appendix: ALS Convergence}

\textbf{Structure:} Block coordinate descent on $(U, V)$ with $M$ partially observed.

\textbf{Each step:} solves a convex subproblem (strongly convex due to regularization $\lambda > 0$ 
\[
U^{k+1} = \arg\min_U f(U, V^k), \quad
V^{k+1} = \arg\min_V f(U^{k+1}, V)
\]

\textbf{Monotonicity:}
\[
f(U^{k+1}, V^{k+1}) \leq f(U^{k+1}, V^k) \leq f(U^k, V^k)
\]

\textbf{Convergence:}
\begin{itemize}
    \item Objective decreases monotonically at every block update
    \item Converges to a stationary point, no global optimality guarantee
\end{itemize}

Sublinear convergence ($O(1/k)$) to a stationary point under mild conditions; can be linear if the objective is strongly convex in each block. 

\end{frame}

\begin{frame}{Appendix: Stationary Point Characterization}

\textbf{Stationary point condition:} First-order necessary conditions for a local minimum in smooth unconstrained optimization.
\[
\nabla_U f(U^*, V^*) = 0, \quad \nabla_V f(U^*, V^*) = 0
\]

\textbf{Explicit gradients:} regularized least-squares with regularization term $\lambda > 0$
\[
(UV^T - M)V + \lambda U = 0
\]
\[
(UV^T - M)^T U + \lambda V = 0
\]

\textbf{Interpretation:}
\begin{itemize}
    \item All three methods (GD, L-BFGS, ALS) converge to solutions satisfying these conditions.
    \item Solutions may be local minima, saddle points, or (rarely) local maxima.
    \item No method guarantees global optimality for this non-convex problem.
\end{itemize}

\end{frame}

\begin{frame}{Appendix: Dense Gradient Derivation}
\[
f(U,V)=\frac12\|UV^\top-M\|_F^2+\frac{\lambda}{2}\left(\|U\|_F^2+\|V\|_F^2\right)
\]
Let \(E=UV^\top-M\). Then
\[
\nabla_U \frac12\|E\|_F^2 = EV=(UV^\top-M)V,
\qquad
\nabla_V \frac12\|E\|_F^2 = E^\top U=(UV^\top-M)^\top U.
\]
Also,
\[
\nabla_U \frac{\lambda}{2}\|U\|_F^2=\lambda U,
\qquad
\nabla_V \frac{\lambda}{2}\|V\|_F^2=\lambda V.
\]
So
\[
\nabla_U f=(UV^\top-M)V+\lambda U,
\qquad
\nabla_V f=(UV^\top-M)^\top U+\lambda V.
\]
\end{frame}

\begin{frame}{Appendix: Sparse Gradient Derivation}
\[
f(U,V)=\frac12\sum_{(i,j)\in\Omega}(u_i^\top v_j-M_{ij})^2
+\frac{\lambda}{2}\left(\|U\|_F^2+\|V\|_F^2\right)
\]

For fixed \(u_i\),
\[
\nabla_{u_i} f
=
\sum_{j:(i,j)\in\Omega}
\frac{\partial}{\partial u_i}
\frac12 (u_i^\top v_j-M_{ij})^2
+\lambda u_i
=
\sum_{j:(i,j)\in\Omega}(u_i^\top v_j-M_{ij})v_j+\lambda u_i.
\]

For fixed \(v_j\),
\[
\nabla_{v_j} f
=
\sum_{i:(i,j)\in\Omega}
\frac{\partial}{\partial v_j}
\frac12 (u_i^\top v_j-M_{ij})^2
+\lambda v_j
=
\sum_{i:(i,j)\in\Omega}(u_i^\top v_j-M_{ij})u_i+\lambda v_j.
\]
\end{frame}


\begin{frame}{Appendix: Full BFGS Update}
Define
\[
\rho_k=\frac{1}{y_k^\top s_k}.
\]

The inverse-BFGS update is
\[
H_{k+1}
=
(I-\rho_k s_k y_k^\top)\,H_k\,(I-\rho_k y_k s_k^\top)
+\rho_k s_k s_k^\top.
\]

This update is designed so that:
\begin{itemize}
\item \(H_{k+1}\) is symmetric,
\item \(H_{k+1}\) stays positive definite if \(y_k^\top s_k>0\),
\item the secant condition holds:
\[
H_{k+1}y_k=s_k.
\]
\end{itemize}

Then the search direction is
\[
p_k=-H_k\nabla f(x_k).
\]
\end{frame}

\begin{frame}{Appendix: Two-Loop Recursion}
Let
\[
g_k=\nabla f(x_k),
\qquad
\rho_i=\frac{1}{y_i^\top s_i}.
\]

We compute \(p_k=-H_k g_k\) as follows.

\textbf{Backward loop:}
\[
q=g_k.
\]
For \(i=k-1,k-2,\dots,k-m\):
\[
\alpha_i=\rho_i s_i^\top q,
\qquad
q\leftarrow q-\alpha_i y_i.
\]

\textbf{Initial scaling:}
\[
H_k^{(0)}=\gamma_k I,
\qquad
\gamma_k=\frac{s_{k-1}^\top y_{k-1}}{y_{k-1}^\top y_{k-1}},
\]
and set
\[
r=H_k^{(0)}q.
\]

\textbf{Forward loop:}
For \(i=k-m,\dots,k-1\):
\[
\beta_i=\rho_i y_i^\top r,
\qquad
r\leftarrow r+s_i(\alpha_i-\beta_i).
\]

Then
\[
p_k=-r.
\]
\end{frame}

\begin{frame}{Appendix: Dense ALS Derivation}
\[
f(U,V)=\frac12\|M-UV^\top\|_F^2+\frac{\lambda}{2}\left(\|U\|_F^2+\|V\|_F^2\right)
\]

Fix \(V\):
\[
\nabla_U f=(UV^\top-M)V+\lambda U=0
\]
\[
UV^\top V-MV+\lambda U=0
\]
\[
U(V^\top V+\lambda I)=MV
\]
\[
U=MV(V^\top V+\lambda I)^{-1}
\]

Fix \(U\):
\[
\nabla_V f=(UV^\top-M)^\top U+\lambda V=0
\]
\[
V(U^\top U+\lambda I)=M^\top U
\]
\[
V=M^\top U(U^\top U+\lambda I)^{-1}
\]
\end{frame}


\begin{frame}{Appendix: Sparse ALS Update for \(u_i\)}
\small
Fix \(V\). For row \(i\), let
\[
\Omega_i=\{j:(i,j)\in\Omega\}.
\]

Then the subproblem is
{\footnotesize
\[
\min_{u_i}\;
\frac12\sum_{j\in\Omega_i}(M_{ij}-u_i^\top v_j)^2
+\frac{\lambda}{2}\|u_i\|_2^2.
\]
}

Set the gradient to zero:
{\footnotesize
\[
\sum_{j\in\Omega_i}(u_i^\top v_j-M_{ij})v_j+\lambda u_i=0.
\]
}

Rearrange:
{\footnotesize
\[
\left(\sum_{j\in\Omega_i} v_jv_j^\top+\lambda I\right)u_i
=
\sum_{j\in\Omega_i} M_{ij}v_j.
\]
}

Therefore,
{\footnotesize
\[
u_i=
\left(\sum_{j\in\Omega_i} v_jv_j^\top+\lambda I\right)^{-1}
\left(\sum_{j\in\Omega_i} M_{ij}v_j\right).
\]
}
\end{frame}


\begin{frame}{Appendix: Sparse ALS Update for \(v_j\)}
\small
Fix \(U\). For column \(j\), let
\[
\Omega^j=\{i:(i,j)\in\Omega\}.
\]

Then the subproblem is
{\footnotesize
\[
\min_{v_j}\;
\frac12\sum_{i\in\Omega^j}(M_{ij}-u_i^\top v_j)^2
+\frac{\lambda}{2}\|v_j\|_2^2.
\]
}

Set the gradient to zero:
{\footnotesize
\[
\sum_{i\in\Omega^j}(u_i^\top v_j-M_{ij})u_i+\lambda v_j=0.
\]
}

Rearrange:
{\footnotesize
\[
\left(\sum_{i\in\Omega^j} u_i u_i^\top+\lambda I\right)v_j
=
\sum_{i\in\Omega^j} M_{ij}u_i.
\]
}

Therefore,
{\footnotesize
\[
v_j=
\left(\sum_{i\in\Omega^j} u_i u_i^\top+\lambda I\right)^{-1}
\left(\sum_{i\in\Omega^j} M_{ij}u_i\right).
\]
}
\end{frame}








\begin{frame}{Appendix: GD Cost Derivation}
    \textbf{Sparse case ($\Omega$ observed):}
    \begin{itemize}
        \item Compute residual $R_{ij} = U_i V_j^T - M_{ij}$ only for $(i,j)\in\Omega$: $O(\text{nnz} \cdot r)$
        \item For each $(i,j)$, update $\nabla_{U_i} \mathrel{+}= R_{ij} V_j$: $O(\text{nnz} \cdot r)$
        \item For each $(i,j)$, update $\nabla_{V_j} \mathrel{+}= R_{ij} U_i$: $O(\text{nnz} \cdot r)$
        \item Add regularization $\lambda U$, $\lambda V$: $O((n+m)r)$
        \item \textbf{Total:} $O(\text{nnz} \cdot r)$
    \end{itemize}

    \vspace{0.2cm}
    \textbf{Dense case ($\Omega =$ all entries):}
    \begin{itemize}
        \item Compute $UV^T$: $O(n m r)$
        \item Subtract $M$: $O(n m)$
        \item Multiply by $V$ (for $\nabla_U$): $O(n m r)$
        \item Multiply by $U$ (for $\nabla_V$): $O(n m r)$
        \item \textbf{Total:} $O(n m r)$
    \end{itemize}
\end{frame}

\begin{frame}{Appendix: L-BFGS Cost Derivation}
    \textbf{Per-iteration cost components:}
    \begin{enumerate}
        \item Gradient computation: same as GD – $O(\text{nnz} \cdot r)$ (sparse) or $O(n m r)$ (dense)
        \item Two-loop recursion for direction $p_k$:
        \[
        \text{Forward: } q = g_k;\; \text{for } i=k-1,\dots,k-m: \alpha_i = \rho_i s_i^T q,\; q = q - \alpha_i y_i
        \]
        \[
        \text{Backward: } r = H_k^0 q;\; \text{for } i=k-m,\dots,k-1: \beta = \rho_i y_i^T r,\; r = r + s_i(\alpha_i - \beta)
        \]
        \item Each loop iteration: $O(r)$ dot products and axpy operations
        \item Total for two-loop recursion: $O(m_{\text{mem}} \cdot r)$
        \item Line search (Armijo/Wolfe): additional gradient evaluations
    \end{enumerate}
    
    \vspace{0.1cm}
    \textbf{Total per iteration:}
    \[
    \text{Sparse: } O(\text{nnz} \cdot r + m r), \qquad
    \text{Dense: } O(n m r + m r)
    \]
\end{frame}

\begin{frame}{Appendix: ALS Cost Derivation}
    \textbf{Update for $U$ (fix $V$):} Solve for each row $u_i$ independently.
    \[
    u_i = \left( \sum_{j \in \Omega_i} V_j V_j^T + \lambda I \right)^{-1} \left( \sum_{j \in \Omega_i} M_{ij} V_j \right)
    \]

    \textbf{Sparse case:}
\begin{itemize}
    \item Build Gram matrices: $O(\text{nnz} \cdot r^2)$
    \item Solve $r \times r$ systems for $U$ and $V$: $O((n+m) r^3)$
    \item \textbf{Total:} $O(\text{nnz} \cdot r^2 + (n+m) r^3)$
\end{itemize}

\vspace{0.15cm}
\textbf{Dense case ($\text{nnz} = nm$):}
\begin{itemize}
    \item Compute $V^TV$: $O(m r^2)$, invert: $O(r^3)$
    \item Compute $MV$: $O(n m r)$, multiply by inverse: $O(n r^2)$
    \item \textbf{Total:} $O(n m r + (n+m) r^2 + r^3)$
\end{itemize}
\end{frame}
